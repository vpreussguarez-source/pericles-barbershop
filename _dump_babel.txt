const timeToMins = t => { const [h,m] = t.split(':').map(Number); return h*60+m; };
const minsToTime = m => `${Math.floor(m/60)}:${m%60===0?'00':'30'}`;
const fmtDuration = mins => {
  if (mins < 60) return `${mins} min`;
  const h = Math.floor(mins/60), m = mins%60;
  return m ? `${h}h ${m}min` : `${h}h`;
};

/** Retorna todos os slots bloqueados por um agendamento (start + duração) */
const getSlotsNeeded = (startTime, durationMins) => {
  const start = timeToMins(startTime);
  if (start + durationMins > 20*60) return []; // ultrapassa horário de fechamento
  const slots = [];
  for (let off = 0; off < durationMins; off += 30) slots.push(minsToTime(start + off));
  return slots;
};

/** Verifica se um horário de início está disponível dado o conjunto de slots ocupados e a duração total */
const isAvailable = (startTime, takenSet, durationMins) => {
  const needed = getSlotsNeeded(startTime, durationMins);
  if (!needed.length) return false; // ultrapassa 20h
  return !needed.some(s => takenSet.has(s));
};

/** Busca slots ocupados no Supabase para atendente + data */
const loadTakenSlots = async (attendantId, date) => {
  if (!_sb) {
    // Modo demo: simula alguns slots ocupados deterministicamente
    const seed = date.getDate() + date.getMonth() * 31;
    const taken = new Set();
    const allSlots = [];
    for (let h=8;h<20;h++) for (let m of [0,30]) { if(h===19&&m===30)continue; allSlots.push(`${h}:${m===0?'00':'30'}`); }
    allSlots.forEach(s => { if((seed*(timeToMins(s)+1)*7)%5===0) taken.add(s); });
    return taken;
  }
  const dateStr = date.toISOString().split('T')[0];
  const { data, error } = await _sb
    .from('bookings')
    .select('slots')
    .eq('attendant_id', attendantId)
    .eq('date', dateStr)
    .neq('status', 'cancelled');
  if (error) { console.error('Supabase read error:', error); return new Set(); }
  const taken = new Set();
  data.forEach(b => b.slots.forEach(s => taken.add(s)));
  return taken;
};

/** Salva o agendamento no Supabase e dispara o webhook */
const saveBooking = async (booking, slots, durationMins, totalPrice, serviceNames) => {
  const dateStr = booking.date.toISOString().split('T')[0];

    attendant_id:   booking.attendant.id,
    attendant_name: booking.attendant.name,
    service_names:  serviceNames,
    date:           dateStr,
    start_time:     booking.time,
    slots,
    duration_mins:  durationMins,
    client_name:    booking.clientName,
    client_phone:   booking.clientPhone,
    total_price:    totalPrice,
    status:         'pending',
  };

  if (!_sb) {
    // Modo demo: simula latência e retorna ID fictício
    await new Promise(r => setTimeout(r, 900));
    const demoId = 'demo-' + Math.random().toString(36).slice(2,10).toUpperCase();
    // Dispara webhook mesmo no demo (se configurado)
    triggerWebhook({ ...payload, id: demoId });
    return { ...payload, id: demoId };
  }

  const { data, error } = await _sb.from('bookings').insert([payload]).select();
  if (error) throw error;
  const record = data[0];
  triggerWebhook(record);
  return record;
};

/** Dispara webhook Make / n8n (não bloqueante) */

    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(record),
  }).catch(e => console.warn('Webhook error:', e));
};

const MONTHS_PT  = ['Janeiro','Fevereiro','Março','Abril','Maio','Junho','Julho','Agosto','Setembro','Outubro','Novembro','Dezembro'];
const WEEKDAYS_PT= ['Dom','Seg','Ter','Qua','Qui','Sex','Sáb'];

  if (!d) return '';

// COMPONENTE — Calendar
// ══════════════════════════════════════════════════════════════════

function Calendar({ value, onChange }) {
  const today = new Date(); today.setHours(0,0,0,0);
  const [viewYear, setViewYear] = useState(today.getFullYear());
  const [viewMonth, setViewMonth] = useState(today.getMonth());
  const firstDay     = new Date(viewYear, viewMonth, 1).getDay();
  const daysInMonth  = new Date(viewYear, viewMonth + 1, 0).getDate();
  const prevMonth = () => viewMonth===0 ? (setViewMonth(11),setViewYear(y=>y-1)) : setViewMonth(m=>m-1);
  const nextMonth = () => viewMonth===11? (setViewMonth(0), setViewYear(y=>y+1)) : setViewMonth(m=>m+1);
  const cells = [];
  for (let i=0;i<firstDay;i++) cells.push(null);
  for (let d=1;d<=daysInMonth;d++) cells.push(d);
  return (
    <div className="calendar-wrap">
      <div className="cal-header">
        <button className="cal-nav" onClick={prevMonth}>‹</button>
        <span className="cal-title">{MONTHS_PT[viewMonth]} {viewYear}</span>
        <button className="cal-nav" onClick={nextMonth}>›</button>
      </div>
      <div className="cal-weekdays">{WEEKDAYS_PT.map(w=><div key={w} className="cal-weekday">{w}</div>)}</div>
      <div className="cal-days">
        {cells.map((d,i) => {
          if (!d) return <div key={`e${i}`} className="cal-day empty"/>;
          const date     = new Date(viewYear, viewMonth, d);
          const disabled = date.getDay()===0 || date<today;
          const isToday  = date.getTime()===today.getTime();
          const isSel    = value && value.getTime()===date.getTime();
          return (
            <div key={d}
              className={`cal-day${disabled?' disabled':''}${isToday?' today':''}${isSel?' selected':''}`}
              onClick={() => !disabled && onChange(date)}
            >{d}</div>
          );
        })}
      </div>
    </div>
  );
}

// COMPONENTE — TimeSlots
// Carrega horários reais do Supabase e bloqueia slots considerando
// a duração total do agendamento
// ══════════════════════════════════════════════════════════════════

function TimeSlots({ attendantId, date, value, onChange, totalDuration }) {
  const [takenSlots, setTakenSlots] = useState(new Set());
  const [loading, setLoading]       = useState(false);

  useEffect(() => {
    if (!date || !attendantId) return;
    setLoading(true);
    setTakenSlots(new Set());
    loadTakenSlots(attendantId, date)
      .then(s => { setTakenSlots(s); setLoading(false); })
      .catch(()=> setLoading(false));
  }, [attendantId, date]);

  const morning=[], afternoon=[];
  for (let h=8;h<20;h++) for (let m of [0,30]) {
    if(h===19&&m===30) continue;
    const lbl=`${h}:${m===0?'00':'30'}`;
    (h<12?morning:afternoon).push(lbl);
  }

  const Slot = ({ t }) => {
    const avail = isAvailable(t, takenSlots, totalDuration);
    const isSel = value===t;
    return (
      <div
        className={`time-slot${!avail?' taken':''}${isSel?' selected':''}`}
        title={!avail ? 'Horário indisponível para a duração do serviço' : ''}
        onClick={() => avail && onChange(t)}
      >{t}</div>
    );
  };

  if (loading) return (
    <div className="loading-row">
      <div className="spinner"/>
      Verificando disponibilidade…
    </div>
  );

  const morningAvail    = morning.filter(t    => isAvailable(t,    takenSlots, totalDuration));
  const afternoonAvail  = afternoon.filter(t  => isAvailable(t,    takenSlots, totalDuration));
  const totalAvail      = morningAvail.length + afternoonAvail.length;

  if (totalAvail === 0) return (
    <div className="no-slots-msg">
      Nenhum horário disponível para esta data com duração de {fmtDuration(totalDuration)}.<br/>
      Tente outra data.
    </div>
  );

  return (
    <div>
      {morningAvail.length > 0 && <>
        <div className="time-section-label">Manhã</div>
        <div className="time-slots">{morning.map(t=><Slot key={t} t={t}/>)}</div>
      </>}
      {afternoonAvail.length > 0 && <>
        <div className="time-section-label">Tarde</div>
        <div className="time-slots">{afternoon.map(t=><Slot key={t} t={t}/>)}</div>
      </>}
    </div>
  );
}

                <div className="btn-row">
                  <button className="btn btn-ghost" onClick={back}>← Voltar</button>
                  <button className="btn btn-primary" disabled={!booking.time} onClick={next}>Continuar →</button>
                </div>
              </div>
            )}

            {/* ─── STEP 5 — Dados do cliente ─── */}
            {step===5 && (
              <div className="form-max">
                <div className="summary-box">
                  <div className="summary-box-title">Resumo do Agendamento</div>
                  {[
                    ['Atendente',    `${booking.attendant?.name} — ${booking.attendant?.role}`],
                    ['Serviço(s)',   serviceNames],
                    ['Data',         fmtDate(booking.date)],
                    ['Horário',      `${booking.time}h`],
                    ['Duração',      fmtDuration(totalDuration)],
                    ['Valor total',  fmt(totalPrice), true],
                  ].map(([label,value,gold])=>(
                    <div key={label} className="summary-row">
                      <span className="summary-row-label" style={{color:'#fff'}}>{label}</span>
                      <span className="summary-row-value" style={gold?{color:'var(--gold)'}:{}}>{value}</span>
                    </div>
                  ))}
                </div>

                {submitErr && <div className="error-box">{submitErr}</div>}

                <div className="form-field">
                  <label className="form-label">Seu Nome</label>
                  <input className="form-input" type="text" placeholder="Como você se chama?"
                    value={booking.clientName} onChange={e=>set('clientName',e.target.value)}/>
                </div>
                <div className="form-field">
                  <label className="form-label">WhatsApp</label>
                  <input className="form-input" type="tel" placeholder="(46) 9 9999-9999"
                    value={booking.clientPhone} onChange={e=>set('clientPhone',e.target.value)}/>
                </div>
                <div className="btn-row">
                  <button className="btn btn-ghost" onClick={back} disabled={submitting}>← Voltar</button>
                  <button className="btn btn-primary"
                    disabled={!booking.clientName.trim()||!booking.clientPhone.trim()||submitting}
                    onClick={handleConfirm}
                  >
                    {submitting
                      ? <><div className="spinner" style={{borderTopColor:'#0a0a0a',borderColor:'rgba(0,0,0,0.2)'}}/> Salvando…</>
                      : 'Confirmar Agendamento →'
                    }
                  </button>
                </div>
              </div>
            )}

            {/* ─── STEP 6 — Confirmação ─── */}
            {step===6 && (
              <div className="confirm-screen">
                <div className="confirm-icon">✓</div>

                <div className="status-badge status-pending">Aguardando confirmação</div>

                <h1 className="confirm-title">Pré-agendamento Registrado!</h1>
                <p className="confirm-subtitle">
                  Obrigado, <strong>{booking.clientName}</strong>! Seu horário foi reservado com status <em>pendente</em>.
                  Clique no botão abaixo para enviar a mensagem de confirmação pelo WhatsApp — a barbearia validará o agendamento em seguida.
                </p>

                {bookingRec?.id && !DEMO_MODE && (
                  <p className="booking-id-row">
                    Código: <span className="booking-id">{bookingRec.id.toString().slice(0,8).toUpperCase()}</span>
                  </p>
                )}
                {DEMO_MODE && (
                  <p className="booking-id-row" style={{color:'var(--gold)',marginTop:'4px'}}>
                    Modo demo — agendamento não salvo no banco de dados.
                  </p>
                )}

                <div className="confirm-card" style={{marginTop:'32px'}}>
                  <div className="confirm-card-header">Detalhes do Agendamento</div>
                  {[
                    {label:'Cliente',    value: booking.clientName},
                    {label:'WhatsApp',   value: booking.clientPhone},
                    {label:'Serviço(s)', value: serviceNames},
                    {label:'Atendente',  value: `${booking.attendant?.name} — ${booking.attendant?.role}`},
                    {label:'Data',       value: fmtDate(booking.date)},
                    {label:'Horário',    value: `${booking.time}h`},
                    {label:'Duração',    value: fmtDuration(totalDuration)},
                    {label:'Valor',      value: fmt(totalPrice), gold:true},
                  ].map(row=>(
                    <div key={row.label} className="confirm-detail">
                      {dot}
                      <div>
                        <span className="confirm-detail-label">{row.label}</span>
                        <span className="confirm-detail-value" style={row.gold?{color:'var(--gold)'}:{}}>{row.value}</span>
                      </div>
                    </div>
                  ))}
                </div>

                <div style={{display:'flex',gap:'16px',flexWrap:'wrap',justifyContent:'center'}}>
                  <a href={buildWhatsApp(bookingRec)} target="_blank" rel="noopener noreferrer">
                    <button className="btn btn-whatsapp">
                      <svg width="18" height="18" viewBox="0 0 24 24" fill="currentColor">
                        <path d="M17.472 14.382c-.297-.149-1.758-.867-2.03-.967-.273-.099-.471-.148-.67.15-.197.297-.767.966-.94 1.164-.173.199-.347.223-.644.075-.297-.15-1.255-.463-2.39-1.475-.883-.788-1.48-1.761-1.653-2.059-.173-.297-.018-.458.13-.606.134-.133.298-.347.446-.52.149-.174.198-.298.298-.497.099-.198.05-.371-.025-.52-.075-.149-.669-1.612-.916-2.207-.242-.579-.487-.5-.669-.51-.173-.008-.371-.01-.57-.01-.198 0-.52.074-.792.372-.272.297-1.04 1.016-1.04 2.479 0 1.462 1.065 2.875 1.213 3.074.149.198 2.096 3.2 5.077 4.487.709.306 1.262.489 1.694.625.712.227 1.36.195 1.871.118.571-.085 1.758-.719 2.006-1.413.248-.694.248-1.289.173-1.413-.074-.124-.272-.198-.57-.347"/>
                      </svg>
                      Confirmar pelo WhatsApp
                    </button>
                  </a>
                  <button className="btn btn-ghost" onClick={()=>{
                    setBooking({attendant:null,services:[],date:null,time:null,clientName:'',clientPhone:''});
                    setBookingRec(null); setSubmitErr(''); go(1);
                  }}>
                    Novo Agendamento
                  </button>
                </div>

                <p style={{marginTop:'32px',fontSize:'12px',color:'var(--text-dim)',textAlign:'center',lineHeight:'1.6'}}>
                  Rua Tocantins, 2396 — Sala 1 — Centro, Pato Branco — PR<br/>
                  Segunda a Sábado · 08h às 20h
                </p>
              </div>
            )}

          </div>
        </div>
      </main>
    </div>
  );
}

ReactDOM.createRoot(document.getElementById('agendamento-root')).render(<App/>);
